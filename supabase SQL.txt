-- Tablas para perfiles vinculados con cuentas de Supabase Auth.
-- Este script se puede ejecutar en un proyecto que ya tenga las tablas originales.

CREATE TABLE IF NOT EXISTS public.doctores (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    nombre_completo TEXT NOT NULL,
    correo_electronico TEXT UNIQUE NOT NULL,
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

CREATE TABLE IF NOT EXISTS public.pacientes (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    auth_user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
    correo_electronico TEXT,
    nombre TEXT NOT NULL,
    edad INT NOT NULL,
    presion_arterial TEXT,
    frecuencia_cardiaca INT,
    saturacion_oxigeno NUMERIC(5,2),
    temperatura_expuesta NUMERIC(4,2),
    temperatura_paciente NUMERIC(4,2),
    doctor_id UUID REFERENCES public.doctores(id) ON DELETE SET NULL,
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

CREATE TABLE IF NOT EXISTS public.lecturas_paciente (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    paciente_id UUID REFERENCES public.pacientes(id) ON DELETE CASCADE,
    presion_sistolica INTEGER,
    presion_diastolica INTEGER,
    ritmo_cardiaco INTEGER,
    temperatura_expuesta NUMERIC(6,2),
    humedad_ambiente NUMERIC(5,2),
    temperatura_paciente NUMERIC(5,2),
    spo2 INTEGER
);

-- Migra la tabla existente sin borrar las lecturas que ya envió el ESP32.
ALTER TABLE public.lecturas_paciente
    ADD COLUMN IF NOT EXISTS paciente_id UUID;

ALTER TABLE public.lecturas_paciente
    ALTER COLUMN temperatura_expuesta TYPE NUMERIC(6,2)
        USING temperatura_expuesta::NUMERIC(6,2),
    ALTER COLUMN humedad_ambiente TYPE NUMERIC(5,2)
        USING humedad_ambiente::NUMERIC(5,2),
    ALTER COLUMN temperatura_paciente TYPE NUMERIC(5,2)
        USING temperatura_paciente::NUMERIC(5,2);

-- En el prototipo actual hay un solo paciente; vincula las lecturas antiguas
-- solo cuando existe exactamente un perfil, sin asignarlas a alguien por error.
UPDATE public.lecturas_paciente
SET paciente_id = (
    SELECT id
    FROM public.pacientes
    LIMIT 1
)
WHERE paciente_id IS NULL
  AND (SELECT count(*) FROM public.pacientes) = 1;

DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1
        FROM pg_constraint
        WHERE conname = 'lecturas_paciente_paciente_id_fkey'
          AND conrelid = 'public.lecturas_paciente'::regclass
    ) THEN
        ALTER TABLE public.lecturas_paciente
            ADD CONSTRAINT lecturas_paciente_paciente_id_fkey
            FOREIGN KEY (paciente_id)
            REFERENCES public.pacientes(id)
            ON DELETE CASCADE;
    END IF;

    IF NOT EXISTS (
        SELECT 1
        FROM pg_constraint
        WHERE conname = 'lecturas_paciente_patient_required'
          AND conrelid = 'public.lecturas_paciente'::regclass
    ) THEN
        -- NOT VALID conserva lecturas antiguas sin paciente, pero exige
        -- paciente_id en cada lectura nueva.
        ALTER TABLE public.lecturas_paciente
            ADD CONSTRAINT lecturas_paciente_patient_required
            CHECK (paciente_id IS NOT NULL) NOT VALID;
    END IF;
END
$$;

CREATE INDEX IF NOT EXISTS lecturas_paciente_patient_latest_idx
    ON public.lecturas_paciente (paciente_id, created_at DESC);

ALTER TABLE public.lecturas_paciente ENABLE ROW LEVEL SECURITY;
REVOKE ALL ON TABLE public.lecturas_paciente FROM anon, authenticated;
GRANT SELECT ON TABLE public.lecturas_paciente TO authenticated;
GRANT INSERT ON TABLE public.lecturas_paciente TO service_role;
GRANT USAGE, SELECT ON SEQUENCE public.lecturas_paciente_id_seq TO service_role;

DROP POLICY IF EXISTS "Patients can view own readings"
    ON public.lecturas_paciente;
CREATE POLICY "Patients can view own readings"
    ON public.lecturas_paciente
    FOR SELECT
    TO authenticated
    USING (
        EXISTS (
            SELECT 1
            FROM public.pacientes
            WHERE public.pacientes.id = lecturas_paciente.paciente_id
              AND public.pacientes.auth_user_id = (SELECT auth.uid())
        )
    );

ALTER TABLE public.pacientes
    ADD COLUMN IF NOT EXISTS auth_user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE;
ALTER TABLE public.pacientes
    ADD COLUMN IF NOT EXISTS correo_electronico TEXT;

-- Supabase Auth se encarga de las contraseñas; elimina la columna local anterior.
ALTER TABLE public.pacientes DROP COLUMN IF EXISTS contrasena;

CREATE UNIQUE INDEX IF NOT EXISTS pacientes_auth_user_id_unique
    ON public.pacientes (auth_user_id)
    WHERE auth_user_id IS NOT NULL;
CREATE UNIQUE INDEX IF NOT EXISTS pacientes_correo_electronico_unique
    ON public.pacientes (correo_electronico)
    WHERE correo_electronico IS NOT NULL;

DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1
        FROM pg_constraint
        WHERE conname = 'pacientes_edad_rango'
          AND conrelid = 'public.pacientes'::regclass
    ) THEN
        ALTER TABLE public.pacientes
            ADD CONSTRAINT pacientes_edad_rango
            CHECK (edad BETWEEN 60 AND 100) NOT VALID;
    END IF;
END
$$;

CREATE OR REPLACE FUNCTION public.handle_new_auth_patient()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
    patient_name TEXT := btrim(NEW.raw_user_meta_data ->> 'patient_name');
    doctor_name TEXT := btrim(NEW.raw_user_meta_data ->> 'doctor_name');
    doctor_email TEXT := lower(btrim(NEW.raw_user_meta_data ->> 'doctor_email'));
    patient_age INTEGER;
    linked_doctor_id UUID;
BEGIN
    IF NEW.email IS NULL OR patient_name = '' OR
       patient_name !~ '^[[:alpha:]]+([ ''-][[:alpha:]]+)*$' THEN
        RAISE EXCEPTION 'Invalid patient email or name';
    END IF;

    IF doctor_name !~ '^[[:alpha:]]+([ ''-][[:alpha:]]+)+$' OR
       doctor_email !~ '^[^[:space:]@]+@gmail\.com$' THEN
        RAISE EXCEPTION 'Invalid doctor name or email';
    END IF;

    IF (NEW.raw_user_meta_data ->> 'age') !~ '^[0-9]{1,3}$' THEN
        RAISE EXCEPTION 'Invalid patient age';
    END IF;

    patient_age := (NEW.raw_user_meta_data ->> 'age')::INTEGER;
    IF patient_age < 60 OR patient_age > 100 THEN
        RAISE EXCEPTION 'Patient age must be between 60 and 100';
    END IF;

    INSERT INTO public.doctores (nombre_completo, correo_electronico)
    VALUES (doctor_name, doctor_email)
    ON CONFLICT (correo_electronico) DO NOTHING;

    SELECT id INTO linked_doctor_id
    FROM public.doctores
    WHERE correo_electronico = doctor_email;

    INSERT INTO public.pacientes (
        auth_user_id,
        correo_electronico,
        nombre,
        edad,
        doctor_id
    )
    VALUES (
        NEW.id,
        lower(NEW.email),
        patient_name,
        patient_age,
        linked_doctor_id
    );

    RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS on_auth_user_created_patient ON auth.users;
CREATE TRIGGER on_auth_user_created_patient
    AFTER INSERT ON auth.users
    FOR EACH ROW
    EXECUTE FUNCTION public.handle_new_auth_patient();

ALTER TABLE public.pacientes ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.doctores ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Patients can view own profile" ON public.pacientes;
CREATE POLICY "Patients can view own profile"
    ON public.pacientes
    FOR SELECT
    TO authenticated
    USING ((SELECT auth.uid()) = auth_user_id);

GRANT SELECT ON TABLE public.pacientes TO authenticated;

-- Mantiene el paciente activo del único ESP32 del prototipo.
CREATE TABLE IF NOT EXISTS public.esp32_sesion_activa (
    singleton BOOLEAN PRIMARY KEY DEFAULT TRUE CHECK (singleton),
    auth_user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
    paciente_id UUID NOT NULL REFERENCES public.pacientes(id) ON DELETE CASCADE,
    heartbeat_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

ALTER TABLE public.esp32_sesion_activa ENABLE ROW LEVEL SECURITY;
REVOKE ALL ON TABLE public.esp32_sesion_activa FROM anon, authenticated;
GRANT SELECT, INSERT, UPDATE, DELETE
    ON TABLE public.esp32_sesion_activa TO service_role;

CREATE OR REPLACE FUNCTION public.start_esp32_session(
    p_auth_user_id UUID,
    p_paciente_id UUID
)
RETURNS BOOLEAN
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
    current_auth_user_id UUID;
    current_heartbeat_at TIMESTAMPTZ;
BEGIN
    PERFORM pg_advisory_xact_lock(72611262001::BIGINT);

    IF NOT EXISTS (
        SELECT 1
        FROM public.pacientes
        WHERE id = p_paciente_id
          AND auth_user_id = p_auth_user_id
    ) THEN
        RAISE EXCEPTION 'Patient profile does not match authenticated user';
    END IF;

    SELECT auth_user_id, heartbeat_at
    INTO current_auth_user_id, current_heartbeat_at
    FROM public.esp32_sesion_activa
    WHERE singleton = TRUE
    FOR UPDATE;

    IF FOUND
       AND current_auth_user_id <> p_auth_user_id
       AND current_heartbeat_at > now() - INTERVAL '2 minutes' THEN
        RETURN FALSE;
    END IF;

    INSERT INTO public.esp32_sesion_activa (
        singleton,
        auth_user_id,
        paciente_id,
        heartbeat_at
    )
    VALUES (TRUE, p_auth_user_id, p_paciente_id, now())
    ON CONFLICT (singleton) DO UPDATE
    SET auth_user_id = EXCLUDED.auth_user_id,
        paciente_id = EXCLUDED.paciente_id,
        heartbeat_at = EXCLUDED.heartbeat_at;

    RETURN TRUE;
END;
$$;

CREATE OR REPLACE FUNCTION public.heartbeat_esp32_session(
    p_auth_user_id UUID
)
RETURNS BOOLEAN
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
BEGIN
    PERFORM pg_advisory_xact_lock(72611262001::BIGINT);

    UPDATE public.esp32_sesion_activa
    SET heartbeat_at = now()
    WHERE singleton = TRUE
      AND auth_user_id = p_auth_user_id;

    RETURN FOUND;
END;
$$;

CREATE OR REPLACE FUNCTION public.end_esp32_session(
    p_auth_user_id UUID
)
RETURNS BOOLEAN
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
BEGIN
    PERFORM pg_advisory_xact_lock(72611262001::BIGINT);

    DELETE FROM public.esp32_sesion_activa
    WHERE singleton = TRUE
      AND auth_user_id = p_auth_user_id;

    RETURN FOUND;
END;
$$;

CREATE OR REPLACE FUNCTION public.insert_esp32_reading(
    p_presion_sistolica INTEGER,
    p_presion_diastolica INTEGER,
    p_ritmo_cardiaco INTEGER,
    p_temperatura_expuesta NUMERIC,
    p_humedad_ambiente NUMERIC,
    p_temperatura_paciente NUMERIC,
    p_spo2 INTEGER
)
RETURNS BIGINT
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
    active_patient_id UUID;
    inserted_reading_id BIGINT;
BEGIN
    PERFORM pg_advisory_xact_lock(72611262001::BIGINT);

    SELECT paciente_id
    INTO active_patient_id
    FROM public.esp32_sesion_activa
    WHERE singleton = TRUE
      AND heartbeat_at > now() - INTERVAL '2 minutes'
    FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'No active patient session for ESP32'
            USING ERRCODE = '55000';
    END IF;

    INSERT INTO public.lecturas_paciente (
        paciente_id,
        presion_sistolica,
        presion_diastolica,
        ritmo_cardiaco,
        temperatura_expuesta,
        humedad_ambiente,
        temperatura_paciente,
        spo2
    )
    VALUES (
        active_patient_id,
        p_presion_sistolica,
        p_presion_diastolica,
        p_ritmo_cardiaco,
        p_temperatura_expuesta,
        p_humedad_ambiente,
        p_temperatura_paciente,
        p_spo2
    )
    RETURNING id INTO inserted_reading_id;

    RETURN inserted_reading_id;
END;
$$;

REVOKE ALL ON FUNCTION public.start_esp32_session(UUID, UUID)
    FROM PUBLIC, anon, authenticated;
REVOKE ALL ON FUNCTION public.heartbeat_esp32_session(UUID)
    FROM PUBLIC, anon, authenticated;
REVOKE ALL ON FUNCTION public.end_esp32_session(UUID)
    FROM PUBLIC, anon, authenticated;
REVOKE ALL ON FUNCTION public.insert_esp32_reading(
    INTEGER, INTEGER, INTEGER, NUMERIC, NUMERIC, NUMERIC, INTEGER
) FROM PUBLIC, anon, authenticated;

GRANT EXECUTE ON FUNCTION public.start_esp32_session(UUID, UUID)
    TO service_role;
GRANT EXECUTE ON FUNCTION public.heartbeat_esp32_session(UUID)
    TO service_role;
GRANT EXECUTE ON FUNCTION public.end_esp32_session(UUID)
    TO service_role;
GRANT EXECUTE ON FUNCTION public.insert_esp32_reading(
    INTEGER, INTEGER, INTEGER, NUMERIC, NUMERIC, NUMERIC, INTEGER
) TO service_role;
